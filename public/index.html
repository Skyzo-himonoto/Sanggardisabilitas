⚡ Kode Firmware ESP32 & Blynk

Hardware C++

Salin dan unggah kode ini ke modul ESP32 pada tongkat pintar Anda untuk menyambungkan sensor ultrasonik, MPU6050 Gyro, GPS, dan Blynk.

ESP32_Tongkat_Pintar.ino

#define BLYNK_TEMPLATE_ID "TMPL_PENYINTAS"
#define BLYNK_TEMPLATE_NAME "TongkatPintar"
#define BLYNK_AUTH_TOKEN "YOUR_BLYNK_AUTH_TOKEN"

#include <WiFi.h>
#include <WiFiClient.h>
#include <BlynkSimpleEsp32.h>
#include <Wire.h>
#include <Adafruit_MPU6050.h>
// Pin konfigurasi 
#define TRIG_PIN 5
#define ECHO_PIN 18
#define BUZZER_PIN 19
#define MOTOR_PIN 23

char auth[] = BLYNK_AUTH_TOKEN;
char ssid[] = "NamaWiFi";
char pass[] = "PasswordWiFi";

Adafruit_MPU6050 mpu;

void setup() {
  Serial.begin(115200);
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(MOTOR_PIN, OUTPUT);

  Blynk.begin(auth, ssid, pass);
  Wire.begin();
  mpu.begin();
}

void loop() {
  Blynk.run();

  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH);
  int distance = duration * 0.034 / 2;

  Blynk.virtualWrite(V1, distance);

  if (distance < 30) {
    digitalWrite(BUZZER_PIN, HIGH);
    digitalWrite(MOTOR_PIN, HIGH);
  } else {
    digitalWrite(BUZZER_PIN, LOW);
    digitalWrite(MOTOR_PIN, LOW);
  }

  delay(200);
}
